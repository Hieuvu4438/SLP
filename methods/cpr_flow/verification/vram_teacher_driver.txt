import os
os.environ.setdefault('CUBLAS_WORKSPACE_CONFIG',':4096:8')
import copy,gc,json,time
from pathlib import Path
import torch
from cprflow.models.teacher import SignTeacher
from cprflow.objectives.semantic import multi_positive_nce
from cprflow.objectives.reductions import per_clip_huber
from cprflow.training.engine import make_optimizer
from cprflow.training.ema import EMA

torch.set_num_threads(2);torch.manual_seed(17);torch.use_deterministic_algorithms(True)
torch.cuda.set_per_process_memory_fraction(.9)
model=SignTeacher().cuda();optimizer=make_optimizer(model,1e-4,.01);ema=EMA(model)

def update(batch,frames):
    model.zero_grad(set_to_none=True)
    x=torch.randn(batch,frames,296,device='cuda')
    lens=torch.full((batch,),frames,device='cuda',dtype=torch.long)
    times=(torch.arange(frames,device='cuda')/30)[None].expand(batch,-1)
    text=torch.randn(batch,512,768,device='cuda')
    tm=torch.ones(batch,512,device='cuda',dtype=torch.bool)
    lang=torch.zeros(batch,device='cuda',dtype=torch.long)
    corruption=torch.zeros(batch,frames,device='cuda',dtype=torch.bool)
    corruption[:,::6]=True;corruption[:,1::6]=True
    with torch.autocast('cuda',dtype=torch.bfloat16):
        clean=model(x,lens,times)
        keys=model.encode_text(text,tm,lang)
        pos=torch.eye(batch,device='cuda',dtype=torch.bool);allowed=torch.ones_like(pos)
        forward,_=multi_positive_nce(clean['global'],keys,pos,allowed)
        backward,_=multi_positive_nce(keys,clean['global'],pos,allowed)
        reconstructed=model(x.masked_fill(corruption[...,None],0),lens,times,corruption)['reconstruction']
        rec=torch.stack([per_clip_huber(p,t,corruption) for p,t in zip(reconstructed.split((60,90,90,56),-1),x.split((60,90,90,56),-1))]).mean(0)
        loss=((forward+backward)*.5+.1*rec).mean()
    loss.backward()
    torch.nn.utils.clip_grad_norm_(model.parameters(),1.)
    optimizer.step();ema.update(model)

update(2,4);torch.cuda.synchronize();torch.cuda.empty_cache()
report={'synthetic':True,'full_real_training':False,'stage':'teacher','frames':1024,'max_text_tokens':512,
        'precision':'bf16','deterministic':True,'memory_cap_fraction':.9,'results':{}}
for batch in (16,32,64):
    print('teacher',batch,flush=True)
    torch.cuda.reset_peak_memory_stats();start=time.perf_counter()
    try:
        update(batch,1024);torch.cuda.synchronize()
        report['results'][str(batch)]={'status':'ok','seconds':time.perf_counter()-start,
            'peak_allocated_gib':torch.cuda.max_memory_allocated()/2**30,
            'peak_reserved_gib':torch.cuda.max_memory_reserved()/2**30}
    except torch.cuda.OutOfMemoryError as exc:
        report['results'][str(batch)]={'status':'OOM_under_90_percent_cap','reason':str(exc)}
    print(json.dumps(report['results'][str(batch)]),flush=True)
    model.zero_grad(set_to_none=True);gc.collect();torch.cuda.empty_cache()
    Path('/tmp/cpr_teacher_memory.json').write_text(json.dumps(report,indent=2))
print('DONE',flush=True)
